-- =============================================================================
-- PROBLEM 1: Find the Nth Highest Salary Per Department
-- =============================================================================
-- "Find the 2nd highest salary in each department"

-- SOLUTION using DENSE_RANK (handles ties correctly)
SELECT department_id, name, salary
FROM (
    SELECT
        department_id,
        name,
        salary,
        DENSE_RANK() OVER (PARTITION BY department_id ORDER BY salary DESC) AS rnk
    FROM employees
) ranked
WHERE rnk = 2;

-- Expected output:
-- | department_id | name    | salary |
-- |---------------|---------|--------|
-- | 1             | Bob     | 85000  |  (tied with Charlie)
-- | 1             | Charlie | 85000  |  (tied with Bob)
-- | 2             | Irene   | 75000  |
-- | 3             | Frank   | 72000  |

-- WHY DENSE_RANK and not RANK?
-- DENSE_RANK: 95000→1, 85000→2, 70000→3 (no gaps, intuitive "2nd highest")
-- RANK:       95000→1, 85000→2, 85000→2, 70000→4 (gap at 3, 2nd still correct here)
-- ROW_NUMBER: 95000→1, 85000→2, 85000→3 (arbitrary tiebreak, might miss one)


-- =============================================================================
-- PROBLEM 2: Employees Earning More Than Their Manager (Self-Join)
-- =============================================================================
-- Classic self-join problem

SELECT
    e.name  AS employee,
    e.salary AS employee_salary,
    m.name  AS manager,
    m.salary AS manager_salary
FROM employees e
JOIN employees m ON e.manager_id = m.id
WHERE e.salary > m.salary;

-- Expected: Grace (90000) earns more than her manager Frank (72000)
-- | employee | employee_salary | manager | manager_salary |
-- |----------|-----------------|---------|----------------|
-- | Grace    | 90000           | Frank   | 72000          |

-- VARIATION: Find employees earning MORE THAN the average salary in their department
SELECT e.name, e.salary, e.department_id, dept_avg.avg_salary
FROM employees e
JOIN (
    SELECT department_id, AVG(salary) AS avg_salary
    FROM employees
    GROUP BY department_id
) dept_avg ON e.department_id = dept_avg.department_id
WHERE e.salary > dept_avg.avg_salary;

-- ALTERNATIVE using window function (more elegant):
SELECT name, salary, department_id, avg_salary
FROM (
    SELECT
        name, salary, department_id,
        AVG(salary) OVER (PARTITION BY department_id) AS avg_salary
    FROM employees
) with_avg
WHERE salary > avg_salary;


-- =============================================================================
-- PROBLEM 3: Running Total of Sales by Month
-- =============================================================================
-- Monthly sales with cumulative running total

SELECT
    DATE_TRUNC('month', order_date) AS month,     -- PostgreSQL
    -- DATE_FORMAT(order_date, '%Y-%m') AS month,  -- MySQL alternative
    SUM(amount) AS monthly_sales,
    SUM(SUM(amount)) OVER (ORDER BY DATE_TRUNC('month', order_date)) AS running_total
FROM orders
GROUP BY DATE_TRUNC('month', order_date)
ORDER BY month;

-- Expected:
-- | month   | monthly_sales | running_total |
-- |---------|---------------|---------------|
-- | 2024-01 | 1999.98       | 1999.98       |
-- | 2024-02 | 1029.98       | 3029.96       |
-- | 2024-03 | 109.98        | 3139.94       |
-- | 2024-04 | 29.99         | 3169.93       |
-- | 2024-05 | 79.99         | 3249.92       |
-- | 2024-06 | 1079.98       | 4329.90       |

-- KEY INSIGHT: SUM(SUM(amount)) OVER (...) is valid!
-- The inner SUM is the GROUP BY aggregate; the outer SUM is the window function.
-- Window functions execute AFTER GROUP BY.

-- VARIATION: Running total per customer
SELECT
    customer_id,
    order_date,
    amount,
    SUM(amount) OVER (PARTITION BY customer_id ORDER BY order_date) AS customer_running_total
FROM orders
ORDER BY customer_id, order_date;


-- =============================================================================
-- PROBLEM 4: Customers Who Ordered Every Product (Relational Division)
-- =============================================================================
-- "Find customers who have ordered ALL products at least once"

-- SOLUTION 1: COUNT comparison (most intuitive)
SELECT c.name
FROM customers c
JOIN orders o ON c.id = o.customer_id
GROUP BY c.id, c.name
HAVING COUNT(DISTINCT o.product_id) = (SELECT COUNT(*) FROM products);

-- Expected: Acme Corp (customer 1 ordered product 1, 2, and 3)

-- SOLUTION 2: NOT EXISTS with double negation (theoretical relational division)
-- "Find customers where there does NOT EXIST a product they have NOT ordered"
SELECT c.name
FROM customers c
WHERE NOT EXISTS (
    SELECT p.id
    FROM products p
    WHERE NOT EXISTS (
        SELECT 1
        FROM orders o
        WHERE o.customer_id = c.id AND o.product_id = p.id
    )
);

-- SOLUTION 3: EXCEPT-based (PostgreSQL)
-- For each customer, check if (all products) EXCEPT (their ordered products) is empty
SELECT c.name
FROM customers c
WHERE NOT EXISTS (
    SELECT id FROM products
    EXCEPT
    SELECT product_id FROM orders WHERE customer_id = c.id
);

-- INTERVIEW TIP: Solution 1 is simplest and most common.
-- Mention Solution 2 to show you understand relational algebra.


-- =============================================================================
-- PROBLEM 5: Pivot Data Using CASE WHEN
-- =============================================================================
-- Convert rows to columns: show each department's employee count and avg salary
-- side by side (pivot/cross-tab)

-- PIVOT: Department stats as columns
SELECT
    'Employee Count' AS metric,
    COUNT(CASE WHEN department_id = 1 THEN 1 END) AS "Engineering",
    COUNT(CASE WHEN department_id = 2 THEN 1 END) AS "Marketing",
    COUNT(CASE WHEN department_id = 3 THEN 1 END) AS "Sales"
FROM employees

UNION ALL

SELECT
    'Avg Salary',
    ROUND(AVG(CASE WHEN department_id = 1 THEN salary END), 0),
    ROUND(AVG(CASE WHEN department_id = 2 THEN salary END), 0),
    ROUND(AVG(CASE WHEN department_id = 3 THEN salary END), 0)
FROM employees;

-- Expected:
-- | metric         | Engineering | Marketing | Sales |
-- |----------------|-------------|-----------|-------|
-- | Employee Count | 4           | 3         | 3     |
-- | Avg Salary     | 83750       | 78333     | 76667 |


-- PRACTICAL PIVOT: Monthly sales per customer
SELECT
    customer_id,
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 1 THEN amount ELSE 0 END) AS "Jan",
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 2 THEN amount ELSE 0 END) AS "Feb",
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 3 THEN amount ELSE 0 END) AS "Mar",
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 4 THEN amount ELSE 0 END) AS "Apr",
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 5 THEN amount ELSE 0 END) AS "May",
    SUM(CASE WHEN EXTRACT(MONTH FROM order_date) = 6 THEN amount ELSE 0 END) AS "Jun",
    SUM(amount) AS "Total"
FROM orders
GROUP BY customer_id
ORDER BY customer_id;

-- BONUS 2: Find the most common salary in each department
SELECT department_id, salary, cnt
FROM (
    SELECT
        department_id,
        salary,
        COUNT(*) AS cnt,
        RANK() OVER (PARTITION BY department_id ORDER BY COUNT(*) DESC) AS rnk
    FROM employees
    GROUP BY department_id, salary
) ranked
WHERE rnk = 1;


-- BONUS 3: Year-over-Year growth (using LAG)
-- If we had multi-year data:
SELECT
    year,
    total_sales,
    LAG(total_sales) OVER (ORDER BY year) AS prev_year_sales,
    ROUND(
        (total_sales - LAG(total_sales) OVER (ORDER BY year))
        / LAG(total_sales) OVER (ORDER BY year) * 100, 1
    ) AS yoy_growth_pct
FROM (
    SELECT EXTRACT(YEAR FROM order_date) AS year, SUM(amount) AS total_sales
    FROM orders
    GROUP BY EXTRACT(YEAR FROM order_date)
) yearly;

-- =============================================================================
-- INTERVIEW TIPS FOR SQL PROBLEMS
-- =============================================================================
--
-- 1. ALWAYS clarify: "Should ties be included?" (affects RANK vs DENSE_RANK vs ROW_NUMBER)
-- 2. ALWAYS clarify: "Are there NULLs to handle?" (affects JOIN type and COALESCE usage)
-- 3. Start with the simplest approach, then optimize if asked.
-- 4. Mention indexing: "I'd add an index on (department_id, salary) for this query."
-- 5. Mention EXPLAIN: "I'd check the execution plan to verify index usage."
-- 6. Window functions are your secret weapon -- they solve most "top N per group" problems.
-- 7. Self-joins solve hierarchical queries (employee-manager, parent-child).
